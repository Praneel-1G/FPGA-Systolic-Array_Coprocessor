`timescale 1ns / 1ps

module top_tpu (
    input wire clk,
    input wire rst,
    input wire rx,
    output wire tx
);

    // --- INTERNAL WIRES ---
    wire rx_empty, rx_rd, tx_full, tx_wr;
    wire [7:0] rx_data, tx_data;
    
    wire load_wt;
    wire [127:0] flat_weights;
    wire [31:0]  flat_data_in;
    wire [71:0]  internal_acc_wire;
    wire [31:0]  flat_ppu_out;

    // 1. THE COMMUNICATOR (Locked to 50 MHz / 115200 baud)
    uart #(
        .DVSR(27) 
    ) my_uart (
        .clk(clk), .rst(rst),
        .rx(rx), .tx(tx),
        .rx_empty(rx_empty), .r_data(rx_data), .rd_uart(rx_rd),
        .tx_full(tx_full), .w_data(tx_data), .wr_uart(tx_wr)
    );

    // 2. THE BRAIN
    tpu_controller my_controller (
        .clk(clk), .rst(rst),
        .rx_empty(rx_empty), .rx_data(rx_data), .rx_rd(rx_rd),
        .tx_full(tx_full), .tx_data(tx_data), .tx_wr(tx_wr),
        .load_wt(load_wt), .flat_weights(flat_weights),
        .flat_data_in(flat_data_in), .flat_ppu_out(flat_ppu_out)
    );

    // 3. THE MUSCLE
    systolic_array my_mmu (
        .clk(clk), .rst(rst), .load_wt(load_wt),
        .flat_weights(flat_weights), .flat_data_in(flat_data_in),
        .flat_acc_out(internal_acc_wire)
    );

    // 4. THE PROCESSOR
    ppu my_ppu (
        .clk(clk), .rst(rst),
        .flat_acc_in(internal_acc_wire),
        .flat_ppu_out(flat_ppu_out)
    );

endmodule




`timescale 1ns / 1ps

module tpu_controller (
    input wire clk,
    input wire rst,
    
    // --- UART RX INTERFACE ---
    input wire rx_empty,
    input wire [7:0] rx_data,
    output reg rx_rd,
    
    // --- UART TX INTERFACE ---
    input wire tx_full,
    output reg tx_wr,
    output reg [7:0] tx_data,
    
    // --- MMU / PPU INTERFACE ---
    output reg load_wt,
    output reg [127:0] flat_weights,
    output reg [31:0] flat_data_in,
    input wire [31:0] flat_ppu_out
);

    localparam S_IDLE       = 3'd0;
    localparam S_GET_WTS    = 3'd1;
    localparam S_GET_DATA   = 3'd2;
    localparam S_BURST_MATH = 3'd3;
    localparam S_TX_RESULTS = 3'd4;
    
    reg [2:0] state;
    reg [7:0] byte_count;
    reg [3:0] math_cycle;
    
    reg [7:0] in_buffer  [0:15]; 
    reg [7:0] out_buffer [0:15];

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            state <= S_IDLE;
            rx_rd <= 0;
            tx_wr <= 0;
            load_wt <= 0;
            byte_count <= 0;
            math_cycle <= 0;
            flat_data_in <= 0;
        end else begin
            rx_rd <= 0;
            tx_wr <= 0;
            load_wt <= 0;
            
            case (state)
                S_IDLE: begin
                    if (!rx_empty) begin
                        rx_rd <= 1;
                        if (rx_data == 8'hAA) begin
                            state <= S_GET_WTS;
                            byte_count <= 0;
                        end else if (rx_data == 8'hBB) begin
                            state <= S_GET_DATA;
                            byte_count <= 0;
                        end
                    end
                end
                
                S_GET_WTS: begin
                    if (!rx_empty && !rx_rd) begin
                        rx_rd <= 1;
                        flat_weights <= {flat_weights[119:0], rx_data};
                        byte_count <= byte_count + 1;
                        
                        if (byte_count == 15) begin
                            load_wt <= 1;
                            state <= S_IDLE;
                        end
                    end
                end
                
                S_GET_DATA: begin
                    if (!rx_empty && !rx_rd) begin
                        rx_rd <= 1;
                        in_buffer[byte_count] <= rx_data;
                        byte_count <= byte_count + 1;
                        
                        if (byte_count == 15) begin
                            state <= S_BURST_MATH;
                            math_cycle <= 0;
                        end
                    end
                end
                
                S_BURST_MATH: begin
                    math_cycle <= math_cycle + 1;
                    
                    // 1. PUSH THE DIAGONAL WAVE INTO LEFT DOORS
                    case (math_cycle)
                        0: flat_data_in <= {8'd0, 8'd0, 8'd0, in_buffer[0]};
                        1: flat_data_in <= {8'd0, 8'd0, in_buffer[4], in_buffer[1]};
                        2: flat_data_in <= {8'd0, in_buffer[8], in_buffer[5], in_buffer[2]};
                        3: flat_data_in <= {in_buffer[12], in_buffer[9], in_buffer[6], in_buffer[3]};
                        4: flat_data_in <= {in_buffer[13], in_buffer[10], in_buffer[7], 8'd0};
                        5: flat_data_in <= {in_buffer[14], in_buffer[11], 8'd0, 8'd0};
                        6: flat_data_in <= {in_buffer[15], 8'd0, 8'd0, 8'd0};
                        default: flat_data_in <= 32'd0;
                    endcase
                    
                    // 2. THE DESKEW BUFFER (With structural transposition mapping)
                    if (math_cycle == 6) begin
                        out_buffer[0]  <= flat_ppu_out[7:0];   
                    end
                    if (math_cycle == 7) begin
                        out_buffer[1]  <= flat_ppu_out[7:0];   
                        out_buffer[4]  <= flat_ppu_out[15:8];  
                    end
                    if (math_cycle == 8) begin
                        out_buffer[2]  <= flat_ppu_out[7:0];   
                        out_buffer[5]  <= flat_ppu_out[15:8];  
                        out_buffer[8]  <= flat_ppu_out[23:16]; 
                    end
                    if (math_cycle == 9) begin
                        out_buffer[3]  <= flat_ppu_out[7:0];   
                        out_buffer[6]  <= flat_ppu_out[15:8];  
                        out_buffer[9]  <= flat_ppu_out[23:16]; 
                        out_buffer[12] <= flat_ppu_out[31:24]; 
                    end
                    if (math_cycle == 10) begin
                        out_buffer[7]  <= flat_ppu_out[15:8];  
                        out_buffer[10] <= flat_ppu_out[23:16]; 
                        out_buffer[13] <= flat_ppu_out[31:24]; 
                    end
                    if (math_cycle == 11) begin
                        out_buffer[11] <= flat_ppu_out[23:16]; 
                        out_buffer[14] <= flat_ppu_out[31:24]; 
                    end
                    if (math_cycle == 12) begin
                        out_buffer[15] <= flat_ppu_out[31:24]; 
                    end
                    if (math_cycle == 13) begin
                        state <= S_TX_RESULTS;
                        byte_count <= 0;
                    end
                end
                
                S_TX_RESULTS: begin
                    if (!tx_full && !tx_wr) begin
                        tx_wr <= 1; 
                        tx_data <= out_buffer[byte_count];
                        byte_count <= byte_count + 1;
                        
                        if (byte_count == 15) begin
                            state <= S_IDLE; 
                        end
                    end
                end
            endcase
        end
    end
endmodule



`timescale 1ns / 1ps

module tb_top_tpu();

    // 1. SYSTEM TIMING PARAMETERS
    localparam CLK_PERIOD = 20;   // 50 MHz Clock -> 20ns period
    localparam BIT_TIME = 8680;   // 115200 Baud -> 8680 ns per bit

    reg clk;
    reg rst;
    reg rx;
    wire tx;

    top_tpu dut (
        .clk(clk), .rst(rst), .rx(rx), .tx(tx)
    );

    initial begin
        clk = 0;
        forever #(CLK_PERIOD / 2) clk = ~clk;
    end
    
    // TASK: Mimics Python sending a byte
    task send_uart_byte(input [7:0] data);
        integer i;
        begin
            rx = 0; // Start bit
            #(BIT_TIME);
            for (i = 0; i < 8; i = i + 1) begin
                rx = data[i]; // 8 Data bits
                #(BIT_TIME);
            end
            rx = 1; // Stop bit
            #(BIT_TIME);
        end
    endtask

    // ========================================================
    // CONCURRENT RECEIVER (Runs in background so we never miss data)
    // ========================================================
    integer cap_idx = 0;
    reg [7:0] captured_data [0:15];
    
    initial begin
        cap_idx = 0;
        forever begin
            @(negedge tx); // Wait for true falling edge of start bit
            #(BIT_TIME / 2); // Shift to center of the "eye"
            
            if (tx !== 0) $display("ERROR: False start bit at %0t!", $time);
            #(BIT_TIME);
            
            for (integer i = 0; i < 8; i = i + 1) begin
                captured_data[cap_idx][i] = tx; // Sample 8 Data bits
                #(BIT_TIME);
            end
            
            cap_idx = cap_idx + 1;
            if (tx == 0) wait(tx == 1); // Ensure we wait for the stop bit to clear
        end
    end

    // ========================================================
    // MAIN TEST SEQUENCE
    // ========================================================
    integer idx;
    reg [7:0] expected_data [0:15];
    int errors = 0;

    initial begin
        rx = 1; 
        rst = 1;
        
        $display("=================================================");
        $display("=== STARTING FULL SYSTEM TPU VERIFICATION     ===");
        $display("=================================================");
        
        $dumpfile("top_tpu_full.vcd");
        $dumpvars(0, tb_top_tpu);

        #(CLK_PERIOD * 10);
        rst = 0;
        #(CLK_PERIOD * 10);

        // --- PHASE 1: LOAD WEIGHTS ---
        $display("[%0t] Sending Opcode 0xAA (LOAD_WEIGHTS)", $time);
        send_uart_byte(8'hAA);
        
        for (idx = 0; idx < 16; idx = idx + 1) begin
            if (idx == 0 || idx == 5 || idx == 10 || idx == 15)
                send_uart_byte(8'h01); // Identity Matrix
            else
                send_uart_byte(8'h00); 
        end

        #(BIT_TIME * 5); 
        
        // --- PHASE 2: SEND PIXEL DATA ---
        $display("[%0t] Sending Opcode 0xBB (COMPUTE)", $time);
        send_uart_byte(8'hBB);
        
        for (idx = 0; idx < 16; idx = idx + 1) begin
            send_uart_byte(idx + 1); // Send 1 through 16
        end

        // --- PHASE 3: CAPTURE RESULTS ---
        $display("[%0t] Waiting for TPU processing...", $time);
        
        // Wait for the background receiver to catch all 16 bytes!
        wait(cap_idx == 16);
        
        for (idx = 0; idx < 16; idx = idx + 1) begin
            expected_data[idx] = idx + 1;
        end
        
        for (idx = 0; idx < 16; idx = idx + 1) begin
            $display("[%0t] RX Byte %0d: %0d | Expected: %0d", 
                     $time, idx, captured_data[idx], expected_data[idx]);
            
            if (captured_data[idx] !== expected_data[idx]) begin
                errors = errors + 1;
                $display(" ---> MISMATCH!");
            end
        end

        $display("=================================================");
        if (errors == 0)
            $display("=== TEST PASSED! The TPU is geometrically perfect. ===");
        else
            $display("=== TEST FAILED with %0d errors.                   ===", errors);
        $display("=================================================");
        
        #(BIT_TIME * 10);
        $finish;
    end

    // Safety timeout
    initial begin
        #(100_000_000); 
        $display("CRITICAL ERROR: Simulation timed out!");
        $finish;
    end

endmodule




`timescale 1ns / 1ps

module systolic_array #(
    parameter D_WIDTH = 9,   // 9-bits to force unsigned pixels to positive signed logic
    parameter W_WIDTH = 8,
    parameter ACC_WIDTH = 18 
)(
    input wire clk, 
    input wire rst, 
    input wire load_wt,

    input wire [127:0] flat_weights,
    input wire [31:0]  flat_data_in,
    output wire [71:0] flat_acc_out
);

    // =========================================================
    // 0. UNPACKING THE FLAT CABLES INTO 2D ARRAYS
    // =========================================================
    wire signed [W_WIDTH-1:0] wt_in [0:15];
    wire signed [D_WIDTH-1:0] data_in [0:3];
    wire signed [ACC_WIDTH-1:0] acc_out [0:3];

    // Unpack 32-bit data cable and zero-pad to 9-bit signed positive
    assign data_in[0] = {1'b0, flat_data_in[7:0]};
    assign data_in[1] = {1'b0, flat_data_in[15:8]};
    assign data_in[2] = {1'b0, flat_data_in[23:16]};
    assign data_in[3] = {1'b0, flat_data_in[31:24]};

    // Unpack 128-bit weight cable into sixteen 8-bit blocks
    genvar w;
    generate
        for (w = 0; w < 16; w = w + 1) begin : unpack_wt
            assign wt_in[w] = flat_weights[(w*8)+7 : (w*8)];
        end
    endgenerate

    assign flat_acc_out = {acc_out[3], acc_out[2], acc_out[1], acc_out[0]};

    // =========================================================
    // 1. CREATING THE WIRE GRID 
    // =========================================================
    wire signed [D_WIDTH-1:0] data_link [0:3][0:4];
    wire signed [ACC_WIDTH-1:0] acc_link [0:4][0:3]; 

    // =========================================================
    // 2. WIRING THE EDGES TO THE OUTSIDE WORLD
    // =========================================================
    genvar i;
    generate
        for (i = 0; i < 4; i = i + 1) begin : edge_assign
            assign data_link[i][0] = data_in[i];
            assign acc_link[0][i] = 0;         
            assign acc_out[i] = acc_link[4][i]; 
        end
    endgenerate

    // =========================================================
    // 3. STAMPING OUT THE 16 MAC PEs
    // =========================================================
    genvar r, c;
    generate
        for (r = 0; r < 4; r = r + 1) begin : row
            for (c = 0; c < 4; c = c + 1) begin : col
                
                mac_pe #(
                    .D_WIDTH(D_WIDTH), .W_WIDTH(W_WIDTH), .ACC_WIDTH(ACC_WIDTH)
                ) pe_inst (
                    .clk(clk),
                    .rst(rst),
                    .load_wt(load_wt),
                    .wt_in(wt_in[r*4 + c]),         
                    .data_in(data_link[r][c]),      
                    .data_out(data_link[r][c+1]),   
                    .acc_in(acc_link[r][c]),        
                    .acc_out(acc_link[r+1][c])      
                );
            end
        end
    endgenerate

endmodule
